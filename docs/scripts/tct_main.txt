Private Function GetDeviceById( ByVal lId, ByRef bExists )
    Dim dict
    Set dict = XATM_TCT.Item("XATM_Globals").Devices

    If (dict.Exists(lId)) Then
        ' TODO: Verificar a classe
        Set GetDeviceById = dict(lId)
        bExists = True

    End If
End Function

Function BuscaEquipamento(TipoEquipamento, IdEquipamento, byRef Eqp_Flag)
	Eqp_Flag = False

	If (IdEquipamento >= 0) And (IdEquipamento <= 14) Then
		Set PastaDeDados = Application.GetObject("autoDados.[Subestação].Entrada")
	ElseIf IdEquipamento <= 39 Then
		Set PastaDeDados = Application.GetObject("autoDados.[Subestação].Transformador")
	ElseIf (IdEquipamento <= 49) Or (IdEquipamento >= 80) Then
		Set PastaDeDados = Application.GetObject("autoDados.[Subestação].Barra13_8kV")
	Else
		Set PastaDeDados = Application.GetObject("autoDados.[Subestação].Alimentador")
	End If

	For Each Equip in PastaDeDados
		If TypeName(Equip) = TipoEquipamento Then
			If Equip.Id = IdEquipamento Then
				Eqp_Flag = True
				Exit For
			End If	
		End If
	Next
	
	If Eqp_Flag Then
		Set BuscaEquipamento = Equip
	Else
		Set BuscaEquipamento = Nothing
	End If
End Function


Function BuscaAutomatismo(TipoAutomatismo, IdEquipamento, byRef Atm_Flag)
	Atm_Flag = False

	For Each Atm in Application.GetObject("autoDados.Automatismo")
		If TypeName(Atm) = TipoAutomatismo Then
			Select Case TipoAutomatismo
				Case "XATM_TCT"
					IdTransformador = -999
					On Error Resume Next
						IdTransformador = Atm.Transformador.Id
					On Error GoTo 0
					If IsNumeric(IdTransformador) And IsNumeric(IdEquipamento) Then
						If IdTransformador = IdEquipamento Then
							Atm_Flag = True
						End If	
					End If	
				Case "XATM_IAD","XATM_TCD"
					IdDisjuntor = -999
					On Error Resume Next
						IdDisjuntor = Atm.Disjuntor.Id
					On Error GoTo 0
					If IsNumeric(IdDisjuntor) And IsNumeric(IdEquipamento) Then
						If IdDisjuntor = IdEquipamento Then
							Atm_Flag = True
						End If
					End If
				Case "XATM_Paralelismo"
					Select Case XATM_TCT.Transformador.TrafoAdjacentePreSelecionado
						Case 1
							If ((Atm.HabTrafoID100 = True) And (IdEquipamento = 100)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID200 = True) And (IdEquipamento = 200) And (Atm.HabTrafoID100 = True)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID300 = True) And (IdEquipamento = 300) And (Atm.HabTrafoID100 = True)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID400 = True) And (IdEquipamento = 400) And (Atm.HabTrafoID100 = True)) Then
								Atm_Flag = True
							End If
						Case 2
							If ((Atm.HabTrafoID100 = True) And (IdEquipamento = 100) And (Atm.HabTrafoID200 = True)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID200 = True) And (IdEquipamento = 200)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID300 = True) And (IdEquipamento = 300) And (Atm.HabTrafoID200 = True)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID400 = True) And (IdEquipamento = 400) And (Atm.HabTrafoID200 = True)) Then
								Atm_Flag = True
							End If
						Case 3
							If ((Atm.HabTrafoID100 = True) And (IdEquipamento = 100) And (Atm.HabTrafoID300 = True)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID200 = True) And (IdEquipamento = 200) And (Atm.HabTrafoID300 = True)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID300 = True) And (IdEquipamento = 300)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID400 = True) And (IdEquipamento = 400) And (Atm.HabTrafoID300 = True)) Then
								Atm_Flag = True
							End If
						Case 4
							If ((Atm.HabTrafoID100 = True) And (IdEquipamento = 100) And (Atm.HabTrafoID400 = True)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID200 = True) And (IdEquipamento = 200) And (Atm.HabTrafoID400 = True)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID300 = True) And (IdEquipamento = 300) And (Atm.HabTrafoID400 = True)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID400 = True) And (IdEquipamento = 400)) Then
								Atm_Flag = True
							End If
						Case Else
							If ((Atm.HabTrafoID100 = True) And (IdEquipamento = 100)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID200 = True) And (IdEquipamento = 200)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID300 = True) And (IdEquipamento = 300)) Then
								Atm_Flag = True
							ElseIf ((Atm.HabTrafoID400 = True) And (IdEquipamento = 400)) Then
								Atm_Flag = True
							End If
					End Select
				Case Else
					Atm_Flag = True
			End Select		
		End If
		If Atm_Flag Then
			Exit For
		End If
	Next
	
	If Atm_Flag Then
		Set BuscaAutomatismo = Atm
	Else
		Set BuscaAutomatismo = Nothing
	End If
End Function

Sub Main_Passo0()
	' %% Passo 0: Inicialização das variáveis

	' Atualiza dicionario de dispositivos
	XATM_TCT.Item("XATM_Globals").UpdateDeviceDictionary = Not XATM_TCT.Item("XATM_Globals").UpdateDeviceDictionary
	
	If IsEmpty(XATM_TCT.Item("XATM_Globals").Devices) Then
		' Caso vazio, aguarda dicionario ser preenchido
		Exit Sub
	End If

	' Verifica se o automatismo ja foi desenvolvido para este layout
	If Not IsSeLayoutValid() Then
		WriteLog "Passo 0: Tipo de layout da subestação inválido - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	End If

	' Verifica se o transformador de referência do automatismo é válido
	If Not IsReferenceTransformerValid() Then 
		WriteLog "Passo 0: Automatismo sem transformador de referência - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	End If

	' Mapeia sinal de conclusão com sucesso do automatismo de paralelismo, caso exista
	MapParallelismSuccessSignal
	
	Dim triggerTransformerId
	triggerTransformerId = XATM_TCT.Transformador.Id

	Dim adjacentTransformerId
	' adjacentTransformerId = SelectAdjacentTransformerId(triggerTransformerId)
	adjacentTransformerId = ResolveAdjacentTransformer(triggerTransformerId)

	Parent.Item("TrafoAdjSelcId").WriteEx adjacentTransformerId

	If adjacentTransformerId = -1 Then
		WriteLog "Passo 0: Não foi possível selecionar um transformador adjacente válido - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	End If

	If IsLoadExceedingNominal(triggerTransformerId, adjacentTransformerId) Then
		WriteLog "Passo 0: Carga excedendo capacidade nominal do transformador - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	End If

	Dim nextStep
	nextStep = DetermineInitialStep()

	If nextStep = -1 Then
		WriteLog "Passo 0: Não foi possível determinar o próximo passo do automatismo - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	End If
	
	WriteLog "Passo 0: Variáveis inicializadas - Avanço para próximo passo"
	Parent.Item("TimerPasso").Value = 0
	Value = DetermineInitialStep

End Sub

Function DetermineInitialStep()

	DetermineInitialStep = -1

	If Parent.Item("TipoTCT").Value = "Normalização" Then
		DetermineInitialStep = 1
		Exit Function
	End If

	Dim layoutType
	layoutType = GetLayoutType()

	Dim triggerId
	triggerId = XATM_TCT.Transformador.Id

	Dim breakerId

	If layoutType = "3TRDJATSCBTACBT_2SINGLEBB" And triggerId = 200 Then
		
		Dim adjacentId
		adjacentId = Parent.Item("TrafoAdjSelcId").Value

		If adjacentId = 300 Then
			breakerId = 920
		Else
			breakerId = triggerId + 20
		End If

	Else
		breakerId = triggerId + 20
	End If

	' Em caso de layout de barra sem interligador e disjuntor de baixa inicialmente
	' aberto, avança para o passo 5 (abertura da seccionadora de alta)
	Dim breaker, breakerExists
	Set breaker = GetDeviceById(breakerId, breakerExists)	
	
	If Not breakerExists Then
		DetermineInitialStep = -1
		WriteLog "Passo 0: Disjuntor de baixa do transformador de disparo não encontrado - Bloqueio do Automatismo"
		Exit Function
	End If

	If breaker.Item("Dados").Item("Posição").Value = 1 Then
		DetermineInitialStep = 5
	Else
		DetermineInitialStep = 1
	End If

End Function

		IsSeLayoutValid = True
	End If

Function ChooseBestAdjacentTransformer(triggerId)

    ChooseBestAdjacentTransformer = -1

    Dim candidates
    Set candidates = GetAdjacentCandidates(triggerId)

    Dim transformer
    Set transformer = XATM_TCT.Transformador

    ' =========================
    ' PRÉ-SELEÇÃO
    ' =========================
    Dim preselectedId
    preselectedId = transformer.TrafoAdjacentePreSelecionado * 100

    If preselectedId <> 0 Then

        If IsTransformerEligible(triggerId, preselectedId) Then

            WriteLog "Transformador " & preselectedId & " selecionado por pré-seleção"

            ChooseBestAdjacentTransformer = preselectedId

        Else

            WriteLog "Passo 0: Transformador pré-selecionado (ID " & _
                     preselectedId & ") não elegível - Bloqueio do Automatismo"

        End If

        Exit Function

    End If

    ' =========================
    ' FILTRO DE ELEGIBILIDADE
    ' =========================
    Dim eligible
    Set eligible = CreateObject("Scripting.Dictionary")

    Dim adjacentId
    For Each adjacentId In candidates.Keys()

        If IsTransformerEligible(triggerId, adjacentId) Then
            eligible(adjacentId) = True
        End If

    Next

    If eligible.Count = 0 Then
        WriteLog "Nenhum transformador elegível"
        Exit Function
    End If

    ' =========================
    ' SCORE
    ' =========================
    Dim bestId, bestScore
    bestScore = -1E+20

    For Each adjacentId In eligible.Keys()

        Dim score
        score = 0

        ' =========================
        ' CAMINHO PREFERENCIAL
        ' =========================
        If IsPreferredAdjacentPathClosed(triggerId, adjacentId) Then
            score = score + 3000
        End If

        ' =========================
        ' TIE FECHADO
        ' =========================
        If IsTieClosed(triggerId, adjacentId) Then
            score = score + 2000
        End If

        ' =========================
        ' PARALELISMO
        ' =========================
        If IsTransformerParallelPossible(adjacentId) Then
            score = score + 1000
        End If

        ' =========================
        ' CAPACIDADE DISPONÍVEL
        ' =========================
        Dim availableCapacity
        availableCapacity = GetAvailableLoadCapacity(adjacentId)
        
        score = score + availableCapacity

        ' =========================
        ' MARGEM
        ' =========================
        Dim margin
        margin = GetTransformerMargin(adjacentId)

        score = score + margin

        WriteLog "Trafo " & adjacentId & _
                " | Score=" & score & _
                " | Tie=" & IsTieClosed(triggerId, adjacentId) & _
                " | CaminhoPref=" & IsPreferredAdjacentPathClosed(triggerId, adjacentId) & _
                " | Paralelo=" & IsTransformerParallelPossible(adjacentId) & _
                " | CapacidadeDisp=" & availableCapacity

        If score > bestScore Then
            bestScore = score
            bestId = adjacentId
        End If

    Next

    If bestId <> 0 Then
        ChooseBestAdjacentTransformer = bestId
    End If

End Function

Function GetTransformerMargin(transformerId)

    Dim transformer, exists
    Set transformer = GetDeviceById(transformerId, exists)

    If Not exists Then
        GetTransformerMargin = -1E+20
        Exit Function
    End If

    Dim currentLoad
    currentLoad = CalculateBusbarApparentPower(transformerId)

    GetTransformerMargin = transformer.PotenciaNominal - currentLoad

End Function

Function IsTieClosed(triggerId, adjacentId)

    IsTieClosed = False

    If IsSingleBusbar(triggerId, adjacentId) Then
        IsTieClosed = True
        Exit Function
    End If

    Dim layoutType
    layoutType = GetLayoutType()

    Dim tieId
    tieId = -1

    ' =========================
    ' LAYOUT EM ANEL (3BR3DJ)
    ' =========================
    If layoutType = "3TRSC_3BR3DJ" Or layoutType = "3TRSCBT_3BR3DJ" Then
        
        ' Relações no anel:
        ' 100-200 ? 700
        ' 200-300 ? 710
        ' 100-300 ? 720

        If (triggerId = 100 And adjacentId = 200) Or _
           (triggerId = 200 And adjacentId = 100) Then

            tieId = 700

        ElseIf (triggerId = 200 And adjacentId = 300) Or _
               (triggerId = 300 And adjacentId = 200) Then

            tieId = 710

        ElseIf (triggerId = 100 And adjacentId = 300) Or _
               (triggerId = 300 And adjacentId = 100) Then

            tieId = 720

        End If

    Else
        ' =========================
        ' DEFAULT
        ' =========================
        If triggerId = 100 Or triggerId = 200 Then
            tieId = 700
        ElseIf triggerId = 300 Or triggerId = 400 Then
            tieId = 720
        End If
    End If

    If tieId = -1 Then Exit Function

    Dim tie, tieExists
    Set tie = GetDeviceById(tieId, tieExists)

    If Not tieExists Then Exit Function

    IsTieClosed = (tie.Item("Dados").Item("Posição").Value = 2)

End Function

Function IsPreferredAdjacentPathClosed(triggerId, adjacentId)

    IsPreferredAdjacentPathClosed = False

    Dim layoutType
    layoutType = GetLayoutType()

    ' =========================
    ' LAYOUT ESPECIAL: 2SINGLEBB
    ' =========================
    If layoutType <> "3TRDJATSCBTACBT_2SINGLEBB" Then
        Exit Function
    End If

    ' =========================
    ' Regra aplicável apenas para TR200
    ' =========================
    If triggerId <> 200 Then
        Exit Function
    End If

    Dim breakerId
    breakerId = -1

    ' =========================
    ' Caminhos preferenciais
    ' =========================
    ' TR200 ? TR100 : DJ 220
    ' TR200 ? TR300 : DJ 920
    ' =========================
    If adjacentId = 100 Then
        breakerId = 220

    ElseIf adjacentId = 300 Then
        breakerId = 920

    End If

    If breakerId = -1 Then
        Exit Function
    End If

    Dim breaker, breakerExists
    Set breaker = GetDeviceById(breakerId, breakerExists)

    If Not breakerExists Then
        Exit Function
    End If

    IsPreferredAdjacentPathClosed = (breaker.Item("Dados").Item("Posição").Value = 2)

End Function

Function ResolveAdjacentTransformer(triggerId)

	ResolveAdjacentTransformer = -1

    Dim layoutType
    layoutType = GetLayoutType()

	Dim transformer
    Set transformer = XATM_TCT.Transformador

    Select Case layoutType

        ' =========================
        ' SCORE (ANEL)
        ' =========================
        Case "3TRSC_3BR3DJ", "3TRSCBT_3BR3DJ"

            ResolveAdjacentTransformer = ChooseBestAdjacentTransformer(triggerId)
            Exit Function

        ' =========================
        ' CASO ESPECIAL (2 BARRAS ACOPLADAS)
        ' =========================
        Case "3TRDJATSCBTACBT_2SINGLEBB"
            
			' =========================
			' TR100 ou TR300 ? sempre 200
			' =========================
			If triggerId = 100 Or triggerId = 300 Then
				
				If IsTransformerEligible(triggerId, 200) Then
					ResolveAdjacentTransformer = 200
				End If
				
				Exit Function
			End If

			' =========================
            ' TR200 ? seleção por score
            ' =========================
            If triggerId = 200 Then

                ResolveAdjacentTransformer = ChooseBestAdjacentTransformer(triggerId)

                If ResolveAdjacentTransformer = -1 Then
                    WriteLog "Passo 0: Nenhum transformador adjacente elegível para o transformador de disparo (ID " & triggerId & ")"
                End If

                Exit Function

            End If
			
            Exit Function

        ' =========================
        ' DEFAULT
        ' =========================
        Case Else

            ResolveAdjacentTransformer = GetDefaultAdjacentTransformerId(triggerId)
            Exit Function

    End Select

End Function

Function GetAdjacentCandidates(triggerId)

    Dim layoutType
    layoutType = GetLayoutType()

    Dim candidates
    Set candidates = CreateObject("Scripting.Dictionary")

    ' =========================
    ' LAYOUT EM ANEL (3BR3DJ)
    ' =========================
    If layoutType = "3TRSC_3BR3DJ" Or layoutType = "3TRSCBT_3BR3DJ" Then

        Select Case triggerId
            Case 100
                candidates(200) = True
                candidates(300) = True

            Case 200
                candidates(100) = True
                candidates(300) = True

            Case 300
                candidates(100) = True
                candidates(200) = True
        End Select
    

    ' =========================
    ' LAYOUT ESPECIAL: T200 conectado a duas barras pelo geral de baixa
    ' =========================
    ElseIf layoutType = "3TRDJATSCBTACBT_2SINGLEBB" Then

        If triggerId = 200 Then
            candidates(100) = True
            candidates(300) = True
        Else
            candidates(200) = True
        End If

    ' =========================
    ' DEFAULT
    ' =========================
    Else

        Dim defaultId
        defaultId = GetDefaultAdjacentTransformerId(triggerId)

        If defaultId <> -1 Then
            candidates(defaultId) = True
        End If

    End If

    Set GetAdjacentCandidates = candidates

End Function

Function IsTransformerEligible(triggerId, adjacentId)

    IsTransformerEligible = False

    Dim transformer, transformerExists
    Set transformer = GetDeviceById(adjacentId, transformerExists)

    If Not transformerExists Then
        WriteLog "Passo 0: Transformador adjacente (ID " & adjacentId & ") não encontrado"
        Exit Function
    End If

    If Not transformerExists Then
        WriteLog "Passo 0: Transformador adjacente (ID " & adjacentId & ") não encontrado"
        Exit Function
    End If

    ' =========================
    ' CHAVE DE ALTA DO ADJACENTE
    ' =========================
	If Not IsAdjacentDisconnectorClosedForLiberation(adjacentId) Then
		WriteLog "Passo 0: Seccionadora de alta do transformador " & adjacentId & " não está fechada"
		Exit Function
	End If

	' =========================
	' DISJUNTOR DE ALTA DO ADJACENTE (CASO EXISTA)
	' =========================
	If Not IsAdjacentHighSideBreakerClosedForLiberation(adjacentId) Then
		WriteLog "Passo 0: Disjuntor de alta do transformador " & adjacentId & " não está fechado"
		Exit Function
	End If

    ' =========================
    ' RELÉ DE BLOQUEIO (86)
    ' =========================
    If IsTransformerLockoutActive(adjacentId) Then
        WriteLog "Passo 0: Relé de bloqueio ativo no transformador de ID = " & adjacentId
        Exit Function
    End If

    ' =========================
    ' MODO TESTE
    ' =========================
    If IsTransformerInTestMode(adjacentId) Then
        WriteLog "Passo 0: Transformador de ID = " & adjacentId & " em modo teste"
        Exit Function
    End If

    ' =========================
    ' OK
    ' =========================
    IsTransformerEligible = True

End Function

Function IsTransformerParallelPossible(adjacentId) 
	
	Dim tct, tctExists 
	Set tct = BuscaAutomatismo("XATM_TCT", adjacentId, tctExists) 
	
	If Not tctExists Then

		WriteLog "Passo 0: Automatismo de TCT não encontrado para o transformador " & adjacentId
		
		IsTransformerParallelPossible = False
		Exit Function 

	End If 
	
	IsTransformerParallelPossible = (tct.PermitirParalelismo And XATM_TCT.PermitirParalelismo) 

End Function

Private Function IsReferenceTransformerValid()

	Dim referenceTransformer
	Set referenceTransformer = Nothing
	On Error Resume Next

Function IsLoadExceedingNominal(triggerId, adjacentId)

    Dim requiredLoad
    requiredLoad = GetRequiredLoadAfterOperation(triggerId, adjacentId)

    Dim availableCapacity
    availableCapacity = GetAvailableCapacityAfterOperation(triggerId, adjacentId)

    WriteLog "Passo 0: Carga requerida = " & requiredLoad & " | Capacidade disponível = " & availableCapacity

    IsLoadExceedingNominal = (requiredLoad > availableCapacity)

End Function
Function GetRequiredLoadAfterOperation(triggerId, adjacentId)

    GetRequiredLoadAfterOperation = 0

    Dim layoutType
    layoutType = GetLayoutType()

    Dim mode
    mode = Parent.Item("TipoTCT").Value

    ' =========================
    ' LAYOUT ESPECIAL: 2SINGLEBB
    ' =========================
    If layoutType = "3TRDJATSCBTACBT_2SINGLEBB" Then

        If mode = "Liberação" Then

            Select Case triggerId

                ' 100 ? 200
                ' 200 ? 100
                Case 100, 200

                    If adjacentId = 100 Or adjacentId = 200 Then
                        GetRequiredLoadAfterOperation = CalculateBusbarApparentPower(100)
                    ElseIf adjacentId = 300 Then
                        GetRequiredLoadAfterOperation = CalculateBusbarApparentPower(300)
                    End If

                ' 300 ? 200
                Case 300

                    GetRequiredLoadAfterOperation = CalculateBusbarApparentPower(300)

            End Select

            Exit Function

        End If

        ' =========================
        ' NORMALIZAÇÃO
        ' =========================
        If mode = "Normalização" Then

            Select Case triggerId

                Case 100
                    GetRequiredLoadAfterOperation = CalculateBusbarApparentPower(100)

                Case 200

                    If adjacentId = 100 Then
                        GetRequiredLoadAfterOperation = CalculateBusbarApparentPower(100)
                    ElseIf adjacentId = 300 Then
                        GetRequiredLoadAfterOperation = CalculateBusbarApparentPower(300)
                    End If

                Case 300
                    GetRequiredLoadAfterOperation = CalculateBusbarApparentPower(300)

            End Select

            Exit Function

        End If

    End If

    ' =========================
    ' COMPORTAMENTO PADRÃO
    ' =========================

    If mode = "Liberação" Then

        GetRequiredLoadAfterOperation = _
            CalculateBusbarApparentPower(triggerId) + _
            CalculateBusbarApparentPower(adjacentId)

        Exit Function

    End If

    If mode = "Normalização" Then

        If IsParallelPermanentEnabledByIds(triggerId, adjacentId) Then

            GetRequiredLoadAfterOperation = _
                CalculateBusbarApparentPower(triggerId) + _
                CalculateBusbarApparentPower(adjacentId)

        Else

            GetRequiredLoadAfterOperation = _
                CalculateBusbarApparentPower(triggerId)

        End If

    End If

End Function

Function GetAvailableCapacityAfterOperation(triggerId, adjacentId)

    GetAvailableCapacityAfterOperation = 0

    Dim triggerTransformer, triggerExists
    Dim adjacentTransformer, adjacentExists

    Set triggerTransformer = GetDeviceById(triggerId, triggerExists)
    Set adjacentTransformer = GetDeviceById(adjacentId, adjacentExists)

    If Not triggerExists Then Exit Function
    If Not adjacentExists Then Exit Function

    Dim mode
    mode = Parent.Item("TipoTCT").Value

    ' =========================
    ' LIBERAÇÃO
    ' =========================
    If mode = "Liberação" Then

        ' O adjacente assumirá a carga
        GetAvailableCapacityAfterOperation = adjacentTransformer.PotenciaNominal
        Exit Function

    End If

    ' =========================
    ' NORMALIZAÇÃO
    ' =========================
    If mode = "Normalização" Then

        If IsParallelPermanentEnabled(triggerTransformer, adjacentTransformer) Then

            GetAvailableCapacityAfterOperation = _
                triggerTransformer.PotenciaNominal + _
                adjacentTransformer.PotenciaNominal

        Else

            ' O transformador normalizado volta a assumir sua carga
            GetAvailableCapacityAfterOperation = _
                triggerTransformer.PotenciaNominal

        End If

    End If

End Function

Function IsParallelPermanentEnabled(triggerTransformer, adjacentTransformer)
    
    IsParallelPermanentOverload = False

    Dim parallelism, parallelismExists
	Set parallelism = BuscaAutomatismo("XATM_Paralelismo", triggerTransformer.Id, parallelismExists)

    If Not parallelismExists Then
        
        WriteLog "Passo 0: Automatismo de Paralelismo (ID " & triggerTransformer.Id & ") não encontrado"
        Exit Function

    End If

    Dim allowTransformerParalleling
	
    Dim tct, tctExists
    Set tct = BuscaAutomatismo("XATM_TCT", adjacentTransformer.Id, tctExists)
    
    If Not tctExists Then
        WriteLog "Passo 0: TCT do transformador adjacente (ID " & adjacentTransformer.Id & ") não encontrado"
        Exit Function
    End If
    
    allowTransformerParalleling = tct.PermitirParalelismo And XATM_TCT.PermitirParalelismo
	
    If Not allowTransformerParalleling Then
        
        Exit Function

    End If

    If parallelism.ParaleloPermanente Then

        IsParallelPermanentEnabled = True

    End If

End Function

Function CalculateBusbarApparentPower(transformerId)

    Dim layoutType
    layoutType = GetLayoutType()

    ' Layout especial: TR1 e TR2 na mesma barra
    If layoutType = "3TRDJATSCBTACBT_2SINGLEBB" Then
        
        Select Case transformerId
        
            Case 100, 200
                CalculateBusbarApparentPower = SumFeedersApparentPowerByTransformerIds(Array(100, 200))
                Exit Function
                
            Case 300
                CalculateBusbarApparentPower = SumFeedersApparentPowerByTransformerIds(300)
                Exit Function
                
        End Select
        
    End If

    ' Comportamento padrão
    CalculateBusbarApparentPower = SumFeedersApparentPowerByTransformerIds(transformerId)

End Function

Function GetBusbarTransformers(transformerId)

    Dim layoutType
    layoutType = GetLayoutType()

    ' ============================================================
    ' Layout especial: 3TRDJATSCBTACBT_2SINGLEBB
    ' TR1 e TR2 estão na mesma barra, TR3 em outra
    ' ============================================================
    If layoutType = "3TRDJATSCBTACBT_2SINGLEBB" Then

        Select Case transformerId

            Case 100, 200
                GetBusbarTransformers = Array(100, 200)
                Exit Function

            Case 300
                GetBusbarTransformers = Array(300)
                Exit Function

        End Select

    End If

    ' ============================================================
    ' Comportamento padrão: cada transformador representa sua própria barra
    ' ============================================================
    GetBusbarTransformers = Array(transformerId)

End Function

Private Function IsSingleBusbar(triggerTransformerId, adjacentTransformerId)
	IsSingleBusbar = False

	Dim busbarType, layoutType
	busbarType = Application.GetObject("autoDados.Automatismo.Aux.TipoBarra").Value
	layoutType = GetLayoutType()

	If busbarType = "1BR" Then
		IsSingleBusbar = True
		Exit Function

	ElseIf layoutType = "4TRDJ_3BR1DJ" Then
		Select Case triggerTransformerId
			Case 100, 200
				IsSingleBusbar = False
			Case 300, 400
				IsSingleBusbar = True
		End Select
		
		Exit Function
	
	ElseIf layoutType = "3TRSC_2BR" Or layoutType = "3TRSCBT_2BR" Then
		IsSingleBusbar = True
		Exit Function

	ElseIf layoutType = "3TRDJATSCBTACBT_2SINGLEBB" Then
		
        IsSingleBusbar = True
		
		Exit Function
	End If

End Function

Function SumFeedersApparentPowerByTransformerIds(ByVal transformerIds)

    Dim feeder
    Dim totalActivePower
    Dim totalReactivePower

    totalActivePower = 0
    totalReactivePower = 0

    Dim transformerIndex
    Dim feederTransformer

	If Not IsArray(transformerIds) Then
		transformerIds = Array(transformerIds)
	End If

    For Each feeder In Application.GetObject("autoDados.[Subestação].Alimentador")

        If TypeName(feeder) = "XATM_Alim" Then

            feederTransformer = feeder.Trafo * 100

            For Each transformerIndex In transformerIds

                If feederTransformer = transformerIndex Then

                    If feeder.Item("Dados").Item("Posição").Value = 2 _
                       And feeder.NaoConsiderarPotencia = False Then

                        If Not IsNull(feeder.Item("Dados").Item("PotAtivaMem").Value) Then
                            totalActivePower = totalActivePower + feeder.Item("Dados").Item("PotAtivaMem").Value
                        End If

                        If Not IsNull(feeder.Item("Dados").Item("PotReativaMem").Value) Then
                            totalReactivePower = totalReactivePower + feeder.Item("Dados").Item("PotReativaMem").Value
                        End If

                    End If

                    Exit For
                End If

            Next

        End If

    Next

    SumFeedersApparentPowerByTransformerIds = Sqr((totalActivePower ^ 2) + (totalReactivePower ^ 2))

End Function

Sub Main_Passo1()
	' %% Passo 1: Inserir mestria
	
	Dim triggerTransformer
	Set triggerTransformer = GetDeviceById(XATM_TCT.Transformador.Id, bExists)
	
	Dim adjacentTransformer, adjacentTransformerExists, adjacentTransformerId
	adjacentTransformerId = Parent.Item("TrafoAdjSelcId").Value
	Set adjacentTransformer = GetDeviceById(adjacentTransformerId, adjacentTransformerExists)

	
	Dim allowTransformerParalleling
	allowTransformerParalleling = XATM_TCT.PermitirParalelismo
	
	If Application.GetObject("autoDados.Automatismo.Aux.TipoBarra").Value = "3BR3DJ" Then
		
		If adjacentTransformer Is Nothing Then
			
			WriteLog "Passo 1: Bays adjacentes ou interligador de barras abertos - Avanço para próximo passo"
			Parent.Item("TimerPasso").Value = 0
			Value = 2

			Exit Sub

		End If
		
		Dim tct, tctExists
		Set tct = BuscaAutomatismo("XATM_TCT", adjacentTransformer.Id, tctExists)
		
		If Not tctExists Then
			WriteLog "Passo 1: Não foi possível selecionar o transformador adjacente - Bloqueio do Automatismo"
			Main_BloqueioGeral
			Exit Sub
		End If
		
		allowTransformerParalleling = tct.PermitirParalelismo And XATM_TCT.PermitirParalelismo

	End If
	
	If Not allowTransformerParalleling Then
		
		WriteLog "Passo 1: TCT configurada para não permitir paralelismo - Avanço para próximo passo"
		Parent.Item("TimerPasso").Value = 0
		Me.Value = 2

		Exit Sub

	End If

	Dim parallelism, parallelismExists
	Set parallelism = BuscaAutomatismo("XATM_Paralelismo", XATM_TCT.Transformador.Id, parallelismExists)
	
	If Not parallelismExists Then
		
		WriteLog "Passo 1: Chamada do paralelo habilitado, porém o automatismo não existe - Bloqueio do automatismo"
		Main_BloqueioGeral
		
		Exit Sub

	End If

	If parallelism.BloqueioGeral Then
		
		WriteLog "Passo 1: Paralelismo com bloqueio geral atuado - Bloqueio do Automatismo"
		Main_BloqueioGeral
		
		Exit Sub

	End If

	If parallelism.AutomatismoEmAndamento Then
		
		Exit Sub

	End If


	' Verifica necessidade da chamada do paralelo
	If ((Not triggerTransformer.Item("Dados").Item("ParalelismoOk").Value) _
			Or (Not adjacentTransformer.Item("Dados").Item("ParalelismoOk").Value) _
			Or ((Not adjacentTransformer.TrafoEmMestre) And (Not triggerTransformer.TrafoEmMestre))) Then
	
		' Chama paralelismo
		InsertMasterCommand parallelism, triggerTransformer, adjacentTransformer

	Else
		
		XATM_TCT.Item("DadosParalelismo").Item("ConcluidoComSucessoFlag").WriteEx False
		WriteLog "Passo 1: Trafos paralelados com taps equivalentes - Avanço para próximo passo"
		Parent.Item("TimerPasso").Value = 0
		Value = 2

	End If

End Sub

Private Sub InsertMasterCommand(parallelism, triggerTransformer, adjacentTransformer)

	If parallelism.AutomatismoEmAndamento Then
		Exit Sub
	End If

	
	If parallelism.BloqueioOperador Then	
		WriteLog "Passo 1: Paralelismo com bloqueio do operador atuado - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	End If
		
	If parallelism.BloqueioGeral Then
		WriteLog "Passo 1: Paralelismo com bloqueio geral atuado - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	End If

	If parallelism.BloqueioAutomatico Then
		WriteLog "Passo 1: Paralelismo com bloqueio automático atuado - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	End If

	If Not parallelism.PreCondicoes Then
		WriteLog "Passo 1: Pré-condições do paralelismo não atentidas - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	End If


	If (adjacentTransformer.TrafoEmMestre = False And triggerTransformer.TrafoEmMestre = False) _
			Or adjacentTransformer.Item("Dados").Item("ParalelismoOk").Value = False _
			Or triggerTransformer.Item("Dados").Item("ParalelismoOk").Value = False Then

		If Parent.Item("TipoTCT").Value = "Liberação" Then
			triggerTransformer.Item("Dados").Item("CmdMestreIndividual").WriteEx "Mestre"
		Else
			adjacentTransformer.Item("Dados").Item("CmdMestreIndividual").WriteEx "Mestre"
		End If

	ElseIf triggerTransformer.Item("Dados").Item("ParalelismoOk").Value = True _
			And adjacentTransformer.Item("Dados").Item("ParalelismoOk").Value = True Then

		WriteLog "Passo 1: Transformadores com taps equivalentes - Avanço para próximo passo"
		If Parent.Item("TipoTCT").Value = "Liberação" Then
			triggerTransformer.Item("Dados").Item("CmdMestreIndividual").WriteEx "Mestre"
		Else
			adjacentTransformer.Item("Dados").Item("CmdMestreIndividual").WriteEx "Mestre"
		End If

	Else
		
		If Parent.Item("TipoTCT").Value = "Liberação" Then
			triggerTransformer.Item("Dados").Item("CmdMestreIndividual").WriteEx "Mestre"
		Else
			adjacentTransformer.Item("Dados").Item("CmdMestreIndividual").WriteEx "Mestre"
		End If

	End If
	
End Sub

Sub Passo2L()
	
	' TODO: Trafo = 3TRSC e 3TRSCBT
	' TODO: Barra = 3BR3DJ
	
	Dim layoutType
	layoutType = GetLayoutType()

	Dim triggerTransformerId
	triggerTransformerId = XATM_TCT.Transformador.Id

	Dim adjacentTransformerId
	adjacentTransformerId = Parent.Item("TrafoAdjSelcId").Value

	Dim parallelismAllowed
	parallelismAllowed = IsAllowedTransformerParalleling()

	If Not parallelismAllowed Then
		
		' ====================
		' Em caso de não permissão de paralelismo, abrir o disjuntor de interligação de barras ou, em caso de barra única, 
		' avança para o próximo passo, já que se espera que o disjuntor de baixa do transformador adjacente esteja aberto.
		' ====================

		Dim sameBus
		sameBus = IsSingleBusbar(triggerTransformerId, adjacentTransformerId)

		Dim tieId

		If sameBus Then
			
			WriteLog "Passo 2: Nenhuma ação necessária para transformadores no mesmo barramento sem permissão de paralelismo - Avanço para próximo passo"
			Parent.Item("TimerPasso").Value = 0
			Value = 3
			
			Exit Sub

		Else

			If layoutType = "3TRDJATSCBTACBT_2SINGLEBB" Then
				
				If adjacentTransformerId = 300 Then
					
					WriteLog "Passo 2: Nenhuma ação necessária quando TR300 selecionado como adjacente sem permissão de paralelismo - Avanço para próximo passo"

					Parent.Item("TimerPasso").Value = 0
					Value = 3
					Exit Sub

				End If
				
			Else
				tieId = 700
			End If

		End If

		Dim tie, tieExists
		Set tie = GetDeviceById(tieId, tieExists)

		If Not tieExists Then
			Main_BloqueioGeral
			WriteLog "Passo 2: Disjuntor de interligação de barras (ID: " & tieId & ") não foi encontrado - Bloqueio do Automatismo"
			Exit Sub
		End If

		If tie.Item("Dados").Item("Posição").Value <> 1 Then
			
			Select Case tie.Item("Dados").Item("CmdEmExecução").Value
				Case 0, 3
					' Comando ainda não executado
					tie.Item("Dados").Item("CmdAbrirFechar").WriteEx 1
					Exit Sub
				Case 1
					' Falha execução comando
					WriteLog "Passo 2: Falha ao abrir o disjuntor de interligação de barras sem permissão de paralelismo - Bloqueio do Automatismo"
					Main_BloqueioGeral
					Exit Sub
				Case 2
					' Comando em execução
					Exit Sub						
			End Select

		End If

		tie.Item("Dados").Item("Reset").WriteEx 1

	End If

	Dim breakerId, disconnectorId

	If layoutType = "3TRDJATSCBTACBT_2SINGLEBB" Then
		
		If adjacentTransformerId = 200 And triggerTransformerId = 300 Then
			breakerId = 920
			disconnectorId = 951
		Else
			breakerId = adjacentTransformerId + 20
			disconnectorId = adjacentTransformerId + 51
		End If
		
	Else
		breakerId = adjacentTransformerId + 20
		disconnectorId = adjacentTransformerId + 51
	End If

	Dim disconnector, disconnectorExists
	Set disconnector = GetDeviceById(disconnectorId, disconnectorExists)
	
	If disconnectorExists Then
	
		' ====================
		' Em caso de seccionadora de baixa existente no transformador adjacente, 
		' fechar antes do disjuntor por motivo de intertravamento no IED
		' ====================
		
		If disconnector.Item("Dados").Item("Posição").Value <> 2 Then
			
			Select Case disconnector.Item("Dados").Item("CmdEmExecução").Value
				Case 0, 3
					' Comando ainda não executado
					disconnector.Item("Dados").Item("CmdAbrirFechar").WriteEx 2
					Exit Sub
				Case 1
					' Falha execução comando
					WriteLog "Passo 2: Falha ao fechar a seccionadora de baixa do transformador adjacente - Bloqueio do Automatismo"
					Main_BloqueioGeral
					Exit Sub
				Case 2
					' Comando em execução
					Exit Sub						
			End Select

		End If

		disconnector.Item("Dados").Item("Reset").WriteEx 1

	End If

	' ====================
	' Por fim, fechar o disjuntor de baixa do transformador adjacente.
	' ====================
	Dim breaker, breakerExists
	Set breaker = GetDeviceById(breakerId, breakerExists)

	If Not breakerExists Then
		Main_BloqueioGeral
		WriteLog "Passo 2: Disjuntor de baixa do transformador adjacente (ID: " & breakerId & ") não foi encontrado - Bloqueio do Automatismo"
		Exit Sub
	End If

	If breaker.Item("Dados").Item("Posição").Value <> 2 Then
		
		Select Case breaker.Item("Dados").Item("CmdEmExecução").Value
			Case 0, 3
				' Comando ainda não executado
				breaker.Item("Dados").Item("CmdAbrirFechar").WriteEx 2
				Exit Sub
			Case 1
				' Falha execução comando
				WriteLog "Passo 2: Falha ao fechar o disjuntor de baixa do transformador adjacente - Bloqueio do Automatismo"
				Main_BloqueioGeral
				Exit Sub
			Case 2
				' Comando em execução
				Exit Sub						
		End Select

	End If

	breaker.Item("Dados").Item("Reset").WriteEx 1

	WriteLog "Passo 2: Disjuntor de baixa do transformador adjacente fechado - Avanço para próximo passo"
	Parent.Item("TimerPasso").Value = 0
	Value = 3
	
End Sub

Sub Passo3L()

	Dim triggerTransformerId
	triggerTransformerId = XATM_TCT.Transformador.Id

	Dim adjacentTransformerId
	adjacentTransformerId = Parent.Item("TrafoAdjSelcId").Value

	Dim layoutType
	layoutType = GetLayoutType()

	' ====================
	' SELEÇÃO DO DISJUNTOR/SECCIONADORA DE BAIXA
	' ====================
	Dim breakerId, disconnectorId

	If layoutType = "3TRDJATSCBTACBT_2SINGLEBB" And triggerTransformerId = 200 Then
		
		Dim adjacentId
		adjacentId = Parent.Item("TrafoAdjSelcId").Value

		If adjacentId = 300 Then
			breakerId = 920
			disconnectorId = 951
		Else
			breakerId = triggerTransformerId + 20
			disconnectorId = triggerTransformerId + 51
		End If
		
	Else
		
		breakerId = triggerTransformerId + 20
		disconnectorId = triggerTransformerId + 51

	End If
	
	Dim breaker, breakerExists
	Set breaker = GetDeviceById(breakerId, breakerExists)

	If Not breakerExists Then

		WriteLog "Passo 3: Disjuntor de baixa do transformador não encontrado - Bloqueio do Automatismo"
		Main_BloqueioGeral
		
		Exit Sub

	End If

	If breaker.Item("Dados").Item("Posição").Value <> 1 Then

		Select Case breaker.Item("Dados").Item("CmdEmExecução").Value
			Case 0, 3
				' Comando ainda não executado
				breaker.Item("Dados").Item("CmdAbrirFechar").WriteEx 1
				Exit Sub
			Case 1
				' Falha execução comando
				WriteLog "Passo 2: Falha ao abrir o disjuntor de baixa do transformador - Bloqueio do Automatismo"
				Main_BloqueioGeral
				Exit Sub
			Case 2
				' Comando em execução
				Exit Sub						
		End Select

	End If

	breaker.Item("Dados").Item("Reset").WriteEx 1


	' ====================
	' ABERTURA DA SECCIONADORA DE BAIXA (CASO EXISTA)
	' ====================
	Dim disconnector, disconnectorExists
	Set disconnector  = GetDeviceById(disconnectorId, disconnectorExists)

	If disconnectorExists Then

		If disconnector.Item("Dados").Item("Posição").Value <> 1 Then

			Select Case disconnector.Item("Dados").Item("CmdEmExecução").Value
				Case 0, 3
					' Comando ainda não executado
					disconnector.Item("Dados").Item("CmdAbrirFechar").WriteEx 1
					Exit Sub
				Case 1

					' ==========
					' TODO: Verificar se melhor continuar e concluir com falha
					' ==========

					' Falha execução comando
					WriteLog "Passo 2: Falha ao abrir a seccionadora de baixa do transformador - Bloqueio do Automatismo"
					Main_BloqueioGeral
					Exit Sub
				Case 2
					' Comando em execução
					Exit Sub						
			End Select

		End If

		disconnector.Item("Dados").Item("Reset").WriteEx 1

	End If

	' ====================
	' Em caso de não permissão de paralelismo, neste passo será necessário fechar o interligador de barras
	' ====================
	Dim parallelismAllowed
	parallelismAllowed = IsAllowedTransformerParalleling()

	If Not parallelismAllowed Then

		Dim tieId

		If layoutType = "3TRDJATSCBTACBT_2SINGLEBB" Then
				
			If triggerTransformerId = 300 Then
				
				' ====================
				' Nesta arquitetura há um disjuntor que sai da baixa do transformador
				' 200 e se conecta diretamente na barra do transformador 300
				' ====================
				tieId = 920
			
			Else
				
				tieId = adjacentTransformerId + 20

			End If

		Else

			Dim sameBus
			sameBus = IsSingleBusbar(triggerTransformerId, adjacentTransformerId)

			If sameBus Then
				
				tieId = adjacentTransformerId + 20

			Else

				tieId = 700

			End If

		End If

		Dim tie, tieExists
		Set tie  = GetDeviceById(tieId, tieExists)

		If Not tieExists Then

			WriteLog "Passo 3: Disjuntor interligador de barras (ID=" & tieId & ") não encontrado - Bloqueio do Automatismo"
			Main_BloqueioGeral
			
			Exit Sub

		End If

		If tie.Item("Dados").Item("Posição").Value <> 2 Then
			
			Select Case tie.Item("Dados").Item("CmdEmExecução").Value
				Case 0, 3
					' Comando ainda não executado
					tie.Item("Dados").Item("CmdAbrirFechar").WriteEx 2
					Exit Sub
				Case 1
					' Falha execução comando
					WriteLog "Passo 2: Falha ao abrir o disjuntor de interligação de barras sem permissão de paralelismo - Bloqueio do Automatismo"
					Main_BloqueioGeral
					Exit Sub
				Case 2
					' Comando em execução
					Exit Sub						
			End Select

		End If

		tie.Item("Dados").Item("Reset").WriteEx 1

	End If

	WriteLog "Passo 3: Disjuntor de baixa do transformador aberto - Avanço para próximo passo"
	Parent.Item("TimerPasso").Value = 0
	Value = 4

End Sub

Sub Main_Passo4()

	'
	'	Normalização:
	'		Com paralelismo: Fechar disjuntor (e seccionadora, se existir) de baixa do próprio transformador
	'		Sem paralelismo: Abrir interligador de barras e fechar disjuntor de baixa do próprio transformador (e seccionadora, se existir)
	'
	'	Liberação: Abrir disjuntor de alta do próprio transformador

'-------------------------------------------------------------------------------
Sub Passo4N()

	Dim triggerTransformerId
	triggerTransformerId = XATM_TCT.Transformador.Id

	Dim adjacentTransformerId
	adjacentTransformerId = Parent.Item("TrafoAdjSelcId").Value

	Dim layoutType
	layoutType = GetLayoutType()

	Dim parallelismAllowed
	parallelismAllowed = IsAllowedTransformerParalleling

	If Not parallelismAllowed Then
		
		Dim sameBus
		sameBus = IsSingleBusbar(triggerTransformerId, adjacentTransformerId)
		
		Dim tieId
		
		If sameBus Then
			
			tieId = adjacentTransformerId + 20

		Else

			tieId = 700

		End If


		Dim tie, tieExists
		Set tie = GetDeviceById(tieId, tieExists)

		If Not tieExists Then
			
			WriteLog "Passo 4: Interligador de barras (ID=" & tieId & ") não encontrado - Bloqueio do Automatismo"
			Main_BloqueioGeral
			Exit Sub
			
		End If
		
		If tie.Item("Dados").Item("Posição").Value <> 1 Then
			
			Select Case tie.Item("Dados").Item("CmdEmExecução").Value
				Case 0, 3
					' Comando ainda não executado
					tie.Item("Dados").Item("CmdAbrirFechar").WriteEx 1
				Case 1 
					' Falha execução comando
					Main_BloqueioGeral
					Exit Sub
				Case 2 
					' Comando em execução
					Exit sub
			End Select

			Exit Sub

		End If
		
	End If

	' ====================
	' SELEÇÃO DO DISJUNTOR/SECCIONADORA DE BAIXA
	' ====================
	Dim breakerId, disconnectorId

	If layoutType = "3TRDJATSCBTACBT_2SINGLEBB" And triggerTransformerId = 200 Then
		
		If adjacentTransformerId = 300 Then
			breakerId = 920
			disconnectorId = 951
		Else
			breakerId = triggerTransformerId + 20
			disconnectorId = triggerTransformerId + 51
		End If
		
	Else
		
		breakerId = triggerTransformerId + 20
		disconnectorId = triggerTransformerId + 51

	End If

	Dim disconnector, disconnectorExists
	Set disconnector = GetDeviceById(disconnectorId, disconnectorExists)

	' ==================
	' ATENÇÃO: Em arquituras com seccionadora de baixa, é importante que ela seja fechada antes do disjuntor, 
	' uma vez que há um intertravamento que impede o fechamento do disjuntor com a seccionadora aberta
	' ==================
	If disconnectorExists Then
		
		If disconnector.Item("Dados").Item("Posição").Value <> 2 Then
			
			Select Case disconnector.Item("Dados").Item("CmdEmExecução").Value
				Case 0, 3
					' Comando ainda não executado
					disconnector.Item("Dados").Item("CmdAbrirFechar").WriteEx 2
				Case 1
					' Falha execução comando
					WriteLog "Passo 4: Falha de fechamento na seccionadora de baixa do transformador - Bloqueio do automatismo"
					Main_BloqueioGeral
				Case 2
					' Comando em execução
			End Select

			Exit Sub
			
		End If

	End If

	Dim breaker, breakerExists
	Set breaker = GetDeviceById(breakerId, breakerExists)

	If Not breakerExists Then
	
		Main_BloqueioGeral
		WriteLog "Passo 4: Disjuntor de baixa do transformador não existe - Bloqueio do Automatismo"
		Exit Sub
		
	End If

	If breaker.Item("Dados").Item("Posição").Value <> 2 Then
		
		Select Case breaker.Item("Dados").Item("CmdEmExecução").Value
			Case 0, 3
				' Comando ainda não executado
				breaker.Item("Dados").Item("CmdAbrirFechar").WriteEx 2
			Case 1
				' Falha execução comando
				Main_BloqueioGeral
				Exit Sub
			Case 2
				' Comando em execução
				Exit Sub
		End Select

		Exit Sub
	End If

	WriteLog "Passo 4: Dispositivo(s) manobrado(s) com sucesso - Avanço para o próximo passo"
	Parent.Item("TimerPasso").Value = 0
	Value = 5

End Sub

Private Sub SendIndividualCommand(paralelism, triggerTransformer, adjacentTransformer)

	If paralelism.BloqueioOperador Then

		WriteLog "Passo 5: Paralelismo com bloqueio do operador atuado - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	
	End If
	
	If paralelism.BloqueioGeral Then

		WriteLog "Passo 5: Paralelismo com bloqueio geral atuado - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	
	End If
	
	If paralelism.BloqueioAutomatico Then

		WriteLog "Passo 5: Paralelismo com bloqueio automático atuado - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	
	End If
	
	If Not paralelism.PreCondicoes Then

		WriteLog "Passo 5: Pré-condições do paralelismo não atentidas - Bloqueio do Automatismo"
		Main_BloqueioGeral
		Exit Sub
	
	End If
	

	If paralelism.ParaleloPermanente Then
		
		WriteLog "Passo 5: Transformador possibilita paralelo permanente - Avanço para próximo passo"
		Value = 6
		Exit Sub

	End If

	If XATM_TCT.Item("DadosParalelismo").Item("ConcluidoComSucessoFlag").Value Then
		
		XATM_TCT.Item("DadosParalelismo").Item("ConcluidoComSucessoFlag").WriteEx False
		
		WriteLog "Passo 5: Mestria retirada - Avanço para próximo passo"
		Value = 6
		Exit Sub

	End If

	If Not paralelism.AutomatismoEmAndamento Then

		triggerTransformer.Item("Dados").Item("CmdMestreIndividual").WriteEx "Individual"
		
	End If
		
End Sub
