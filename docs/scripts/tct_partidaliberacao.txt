<XATM_TCT.DadosTCT.[PartidaLiberaçãoTCT]:[PartidaLiberaçãoTCT_CmdLiberar]()>
Sub [PartidaLiberaçãoTCT_CmdLiberar]()
	'$TipoSub = Disparo
	'$ModoDisparo = "Liberação"
    Dim lTimeStamp
    
    If (Trim(Me.Value) = "") Then
        Exit Sub    ' <---------------------------------------------- Early Exit
    End If

    ' Armazena o TimeStamp para utilizar no WriteEx "" 
    ' sem disparar o script novamente 
    lTimeStamp = Me.TimeStamp

	If (Value = "Liberar") Then
		For Each Atm in Application.GetObject("autoDados.Automatismo")
			Select Case TypeName(Atm)
				Case "XATM_RASET"
					If Atm.AutomatismoEmAndamento = True Then
						XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Não iniciada - Automatismo " & Atm.Name & " em andamento"
						Exit Sub
					End If
				Case "XATM_RASED"
					If Atm.AutomatismoEmAndamento = True Then
						XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Não iniciada - Automatismo " & Atm.Name & " em andamento"
						Exit Sub
					End If
				Case "XATM_TAL"
					If Atm.AutomatismoEmAndamento = True Then
						XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Não iniciada - Automatismo " & Atm.Name & " em andamento"
						Exit Sub
					End If
				Case "XATM_TCD"
					If Atm.AutomatismoEmAndamento = True Then
						If (Application.GetObject("autoDados.Automatismo.Aux.TipoEntrada").Value = "2DJGIBGIS") And (Application.GetObject("autoDados.Automatismo.Aux.TipoTrafo").Value = "2TRDJ") Then
						
						Else
							XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Não iniciada - Automatismo " & Atm.Name & " em andamento"
							Exit Sub
						End If
					End If
				Case "XATM_TCT"
					If Atm.AutomatismoEmAndamento = True Then
						XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Não iniciada - Automatismo " & Atm.Name & " em andamento"
						Exit Sub
					End If
				Case "XATM_TSP"
					If Atm.AutomatismoEmAndamento = True Then
						XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Não iniciada - Automatismo " & Atm.Name & " em andamento"
						Exit Sub
					End If
				Case "XATM_RASEBF"
					If Atm.AutomatismoEmAndamento = True Then
						XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Não iniciada - Automatismo " & Atm.Name & " em andamento"
						Exit Sub
					End If
				Case "XATM_IAD"
					If Atm.AutomatismoEmAndamento = True Then
						XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Não iniciada - Automatismo " & Atm.Name & " em andamento"
						Exit Sub
					End If
				Case "XATM_Paralelismo"
	
			End Select
		Next
	End If
	
	If XATM_TCT.Enable = True Then
		If XATM_TCT.AutomatismoEmAndamento = False Then
			If XATM_TCT.Item("Timers").Item("BloqueioAutomaticoLib").Output = False Then
				If XATM_TCT.BloqueioOperador = False Then
					If XATM_TCT.Item("Timers").Item("PreCondicoesLib").Output = True Then
						If XATM_TCT.BloqueioGeral = False Then
							If Value = "Liberar" Then 
								XATM_TCT.Item("MaquinaEstados").Item("Main").WriteEx 0
								XATM_TCT.Item("MaquinaEstados").Item("TipoTCT").WriteEx "Liberação"
								XATM_TCT.AutomatismoEmAndamento = True
								XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Disparo TCT - Liberação"
							End If
						Else
							XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Liberação não iniciada por intertravamento - Bloqueio geral atuado."
						End If
					Else
						XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Liberação não iniciada por intertravamento - Pré-Condições não atendidas."
					End If
				Else
					XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Liberação não iniciada por intertravamento - Bloqueio operador atuado."
				End If
			Else
				XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Liberação não iniciada por intertravamento - Bloqueio automático atuado."
			End If
		End If
	Else
		XATM_TCT.Item("XATM_LogTCT").Texto = "[" & XATM_TCT.Name & "] - Automatismo Desabilitado."
	End If

	Me.WriteEx "", lTimeStamp
End Sub
